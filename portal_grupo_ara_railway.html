Carregando...

'; listaDiv.style.display = 'block'; try { const response = await fetch(`${API_URL}/api/users/passwords`, { headers: { 'x-user-email': sessionStorage.getItem('userEmail') || localStorage.getItem('userEmail') || 'adm@adm.com.br' } }); if (!response.ok) { throw new Error(`Erro ${response.status}: ${response.statusText}`); } const users = await response.json(); let html = ''; users.forEach(u => { const senhaHash = u.password_hash ? u.password_hash.substring(0, 25) + '...' : 'N/A'; const htmlU = `
${u.name} (${u.email})
Hash: ${senhaHash}
`; html += htmlU; }); if (users.length === 0) { html = '

Nenhum usuário encontrado

'; } listaDiv.innerHTML = html; } catch (err) { console.error('Erro ao carregar senhas:', err); listaDiv.innerHTML = `

Erro: ${err.message}

`; } } else { listaDiv.style.display = 'none'; } } function abrirResetSenha(email) { usuarioParaReset = email; const modal = document.getElementById('modal-reset-senha'); if (modal) { document.getElementById('modal-email').textContent = email; document.getElementById('modal-nova-senha').value = ''; modal.style.display = 'flex'; document.getElementById('modal-nova-senha').focus(); } } function fecharModalReset() { const modal = document.getElementById('modal-reset-senha'); if (modal) { modal.style.display = 'none'; } usuarioParaReset = null; } async function confirmarResetSenha() { if (!usuarioParaReset) { alert('Nenhum usuário selecionado'); return; } const novaSenha = document.getElementById('modal-nova-senha').value; if (!novaSenha || novaSenha.length < 6) { alert('Senha deve ter pelo menos 6 caracteres'); return; } try { const response = await fetch(`${API_URL}/api/users/${usuarioParaReset}/reset-password`, { method: 'PUT', headers: { 'Content-Type': 'application/json', 'x-user-email': sessionStorage.getItem('userEmail') || localStorage.getItem('userEmail') || 'adm@adm.com.br' }, body: JSON.stringify({ new_password: novaSenha }) }); if (!response.ok) { const error = await response.json(); throw new Error(error.error || 'Erro ao resetar senha'); } alert('Senha resetada com sucesso'); fecharModalReset(); // Recarregar lista const listaDiv = document.getElementById('senhas-lista'); if (listaDiv) { listaDiv.innerHTML = ''; listaDiv.style.display = 'none'; } } catch (err) { console.error('Erro ao resetar senha:', err); alert(`Erro: ${err.message}`); } }
